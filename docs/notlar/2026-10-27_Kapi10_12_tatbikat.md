## Komutlar (Mac, `~/Desktop/Claude Portman Çalışması`)

```
Y="$(ls -t ~/Downloads/portman-*.tar.age | head -1)"; echo "$Y"
age -d -i ~/.config/portman/yedek.key "$Y" | tar -xOf - portman.dump | ssh portman-vps 'umask 077; cat > /root/portman_geri.dump'
ssh -n portman-vps 'timeout 300 bash /opt/portman/guncel/vps/yedek_geri_dene.sh /root/portman_geri.dump' > tatbikat_kapi10.txt 2>&1
ssh -n portman-vps 'rm -f /root/portman_geri.dump'
curl -sI https://rehber.portatifev.com | head -1 >> tatbikat_kapi10.txt
ssh -n portman-vps "setsid nohup bash -c 'bash /opt/portman/guncel/vps/kapi12_tatbikat.sh v2026.10.16; echo == çıkış \$?' > /root/kapi12_v2026.10.16.txt 2>&1 < /dev/null &"
# üç dakika sonra:
ssh -n portman-vps 'cat /root/kapi12_v2026.10.16.txt' > tatbikat_kapi12.txt 2>&1
```
